# ThreadSanitizer defers asynchronous signals until a thread enters its
# runtime, so the signal-based world stop (needed for every code or GOT
# patch) cannot work under TSan. With VAJRA_SANITIZE=thread only the parts
# that do not stop threads by signal are registered: decoding, relocation,
# migration (cooperative safe points) and the cooperative-stop tests.
set(VAJRA_TSAN OFF)
if(VAJRA_SANITIZE MATCHES "thread")
  set(VAJRA_TSAN ON)
endif()

function(vajra_test_exe name)
  add_executable(${name} ${ARGN})
  target_link_libraries(${name} PRIVATE vajra vajra_warnings)
  target_include_directories(${name} PRIVATE ${PROJECT_SOURCE_DIR}/src ${CMAKE_CURRENT_SOURCE_DIR})
  if(VAJRA_PLATFORM STREQUAL "windows")
    target_compile_definitions(${name} PRIVATE NOMINMAX)
  endif()
endfunction()

# Import redirection: a test library called through the PLT/GOT (Linux) or the
# IAT (Windows). On Linux it is linked three ways so all GOT layouts are
# covered: lazy binding, full RELRO (-z now) and -fno-plt (GLOB_DAT slots).
add_library(vajra_testlib SHARED testlib.cpp)
if(VAJRA_PLATFORM STREQUAL "linux")
  target_compile_options(vajra_testlib PRIVATE -fno-plt -fno-semantic-interposition)
  vajra_test_exe(test_got_lazy test_got.cpp)
  target_link_libraries(test_got_lazy PRIVATE vajra_testlib)
  target_compile_options(test_got_lazy PRIVATE -fplt)
  target_link_options(test_got_lazy PRIVATE -Wl,-z,lazy -Wl,-z,norelro)
  vajra_test_exe(test_got_now test_got.cpp)
  target_link_libraries(test_got_now PRIVATE vajra_testlib)
  target_compile_options(test_got_now PRIVATE -fplt)
  target_link_options(test_got_now PRIVATE -Wl,-z,now -Wl,-z,relro)
  vajra_test_exe(test_got_noplt test_got.cpp)
  target_link_libraries(test_got_noplt PRIVATE vajra_testlib)
  target_compile_options(test_got_noplt PRIVATE -fno-plt)
  if(NOT VAJRA_TSAN)
    add_test(NAME test_got_lazy COMMAND test_got_lazy)
    add_test(NAME test_got_now COMMAND test_got_now)
    add_test(NAME test_got_noplt COMMAND test_got_noplt)
  endif()
else()
  vajra_test_exe(test_iat test_got.cpp)
  target_link_libraries(test_iat PRIVATE vajra_testlib)
  add_test(NAME test_iat COMMAND test_iat)
endif()

# Agent: loads a patch module over the protocol.

# Decoder cross-check against objdump (skipped when objdump is not installed).
add_executable(vajra-decode-check decode_check.cpp)
target_link_libraries(vajra-decode-check PRIVATE vajra vajra_warnings)
find_program(VAJRA_OBJDUMP objdump)
if(VAJRA_OBJDUMP AND NOT VAJRA_TSAN)
  set(_files $<TARGET_FILE:vajra> $<TARGET_FILE:test_hook>)
  if(VAJRA_PLATFORM STREQUAL "linux")
    foreach(lib libc.so.6 libstdc++.so.6)
      execute_process(COMMAND ${CMAKE_CXX_COMPILER} -print-file-name=${lib}
                      OUTPUT_VARIABLE f OUTPUT_STRIP_TRAILING_WHITESPACE)
      if(EXISTS "${f}")
        get_filename_component(real "${f}" REALPATH)
        list(APPEND _files "${real}")
      endif()
    endforeach()
  else()
    get_filename_component(_bin ${CMAKE_CXX_COMPILER} DIRECTORY)
    foreach(f "$ENV{SystemRoot}/System32/ntdll.dll" "$ENV{SystemRoot}/System32/kernelbase.dll"
              "${_bin}/libstdc++-6.dll")
      if(EXISTS "${f}")
        list(APPEND _files "${f}")
      endif()
    endforeach()
  endif()
  add_test(NAME decoder_vs_objdump COMMAND vajra-decode-check ${VAJRA_OBJDUMP} ${_files})
  set_tests_properties(decoder_vs_objdump PROPERTIES TIMEOUT 900)
endif()

# Windows: the MinGW runtime DLLs live next to the compiler.
if(VAJRA_PLATFORM STREQUAL "windows")
  get_filename_component(_bin ${CMAKE_CXX_COMPILER} DIRECTORY)
  get_property(_tests DIRECTORY PROPERTY TESTS)
  foreach(t ${_tests})
    if(CMAKE_VERSION VERSION_GREATER_EQUAL 3.22)
      set_tests_properties(${t} PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${_bin}")
    endif()
  endforeach()
endif()
get_property(_tests DIRECTORY PROPERTY TESTS)
foreach(t ${_tests})
  get_test_property(${t} TIMEOUT _to)
  if(NOT _to)
    set_tests_properties(${t} PROPERTIES TIMEOUT 300)
  endif()
endforeach()
